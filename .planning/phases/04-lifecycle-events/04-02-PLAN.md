   ```typescript
   export async function createInvoice(data: CreateInvoiceInput): Promise<InvoiceWithDetails> {
     const result = await fetcher<InvoiceWithDetails>('/api/invoices', {
       method: 'POST',
       body: JSON.stringify(data),
     })
     analytics.invoiceCreated({
       invoice_id: result.id,
       client_id: result.client_id,
       amount: result.total,
       currency: result.currency,
       line_item_count: result.line_items.length,
     })
     return result
   }
   ```
   Track AFTER `fetcher` resolves successfully. If `fetcher` throws, the analytics call is never reached.

     analytics.invoiceSent({
       invoice_id: result.id,
       amount: result.total,
       currency: result.currency,
     })
     return result
